# CODEX RULES — ПРОЕКТ "ВСЁПОНЯТНО"

Версия: 1.0
Проект: PWA + WordPress + Directus + 3D engine
Среда: PROD (работаем напрямую на сервере)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
ОБЩИЕ ПРАВИЛА ОТВЕТА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Summary писать строго НА РУССКОМ языке.
2. Никаких diff-патчей.
3. Никаких сокращений "...".
4. Никаких служебных символов (␍␊ и т.д.).
5. Никаких CRLF (\r\n).
6. Использовать только Unix LF (\n).
7. Если файл изменяется — выводить его ПОЛНОСТЬЮ.
8. Ответ должен содержать только чистый текст файла.
9. Не вставлять пояснения внутрь файла.
10. Комментарии к изменениям — отдельно от кода.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
АРХИТЕКТУРНЫЕ ОГРАНИЧЕНИЯ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ ЗАПРЕЩЕНО менять без прямого указания:

- mu-plugin proxy (vp-proxy-directus.php)
- protected token flow
- /scan
- canonical fix
- глобальные WordPress hooks
- service-worker.js
- manifest.json
- nginx конфигурации
- Docker конфигурации
- ENV файлы

❌ Нельзя:
- удалять существующую рабочую логику
- менять API контракт без явного указания
- трогать Directus schema без запроса
- изменять безопасность авторизации

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
СТИЛЬ КОДА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PHP:
- строгие отступы 2 пробела
- без closing ?> в конце файла
- проверки function_exists перед объявлением функций

JS:
- IIFE или DOMContentLoaded
- без глобальных переменных
- без зависимости от window кроме VP_* namespace
- без inline eval

CSS:
- использовать дизайн-токены
- не ломать существующие классы
- не менять базовую сетку темы

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
ПРАВИЛА ИЗОЛЯЦИИ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Новые функции должны быть:

- изолированными
- минимальными
- не влияющими на существующие страницы

Новые скрипты подключать:
- только через enqueue
- только на нужных страницах
- без глобального подключения

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3D ENGINE ПРАВИЛА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

- model-viewer подключать только как <script type="module">
- не подключать CDN
- не дублировать подключение
- ждать customElements.whenDefined('model-viewer')
- не ломать protected file flow

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PWA ПРАВИЛА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

- не менять service worker стратегию кэширования
- не удалять version-busting
- не ломать оффлайн-режим
- не менять manifest

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

2. Затем полный файл.
3. Без обёрток ```.
4. Без markdown.
5. Без пояснений внутри файла.
