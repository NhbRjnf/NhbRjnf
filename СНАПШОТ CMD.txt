cd /opt/vseponyatno/docker

# 1) Убедись, что directus контейнер жив
docker compose ps directus

# 2) Сними schema snapshot изнутри контейнера
docker compose exec -T directus npx directus schema snapshot /directus/schema.json

# 3) Вытащи файл на хост в /opt/vseponyatno/schema.json
docker compose cp directus:/directus/schema.json /opt/vseponyatno/schema.json

# 4) Проверь
ls -la /opt/vseponyatno/schema.json
